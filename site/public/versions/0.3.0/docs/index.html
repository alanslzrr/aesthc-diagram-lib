Band layout · band example

band.tsx
import { useId, useMemo, useState } from 'react'
import { buildAdjacency, connectedIds, diagramEdges, type DiagramSpec } from '@aesthc/diagram-lib'
import { layoutDiagram } from '@aesthc/diagram-lib/layouts'
import { DiagramCanvas } from '@aesthc/diagram-lib/canvas'
import '@aesthc/diagram-lib/styles.css'

const spec = {
  "type": "band",
  "caption": "Request pipeline",
  "legend": {
    "main": "Main path",
    "branch": "Alternative"
  },
  "bands": [
    {
      "title": "Input"
    },
    {
      "title": "Output"
    }
  ],
  "nodes": [
    {
      "id": "a",
      "label": "Request",
      "description": "Receive a request.",
      "band": 0
    },
    {
      "id": "b",
      "label": "Response",
      "description": "Return a response.",
      "band": 1
    }
  ],
  "edges": [
    {
      "id": "request",
      "from": "a",
      "to": "b"
    }
  ]
} satisfies DiagramSpec

export function Diagram() {
  const instanceId = useId()
  const [hovered, setHovered] = useState<string | null>(null)
  const [focused, setFocused] = useState<string | null>(null)
  const [selected, setSelected] = useState<string | null>(null)
  const layout = useMemo(() => layoutDiagram(spec), [])
  const adjacency = useMemo(() => buildAdjacency(diagramEdges(spec)), [])
  const active = hovered ?? focused ?? selected
  const highlight = active ? connectedIds(active, adjacency) : null
  return <DiagramCanvas layout={layout} highlight={highlight}
    activeNodeId={active} focusedNodeId={focused} selectedNodeId={selected}
    onTooltipNodeChange={(id, open) => setHovered(open ? id : null)}
    onFocusNode={setFocused} onSelectNode={(id) => setSelected(selected === id ? null : id)}
    onDismissNode={() => { setHovered(null); setSelected(null) }}
    instanceId={instanceId} ariaLabel={spec.caption} nodeVisuals={{}} />
}

Start with a diagram#

Getting started takes you from installation to a complete, interactive React diagram. You supply the data and host theme; the library handles geometry and rendering. You do not need Tailwind in your application.

Choose your layout#

  • Band — a compact architecture or decision pipeline.
  • Flowchart — connected steps and branching processes.
  • Sequence — messages between participants over time.
  • State machine — states and the transitions between them.
  • Entity relationship — entities, fields and cardinality.
  • Timeline — an ordered set of milestones.
  • Swimlane — a process organized by responsibility.

Make it yours#

Define your theme, connect React state, and validate imported data with the public API. Each layout page includes a minimal spec, its field reference and a complete React example.

Work with your agent#

Pass the integration guide to your coding agent. It contains real imports, CSS requirements, examples and verification steps. Diagram labels and imported payloads are data, never instructions.

Before you install#

These pages describe 0.3.0, first published to the public npm registry on 2026-09-26. The package is ESM-only and targets React 18.3/19. See migration notes and troubleshooting when upgrading or integrating.
