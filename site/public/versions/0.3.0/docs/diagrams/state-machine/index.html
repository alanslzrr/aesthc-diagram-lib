State machine diagrams

Version 0.3.0. The minimal spec below is validated and its complete React example is compiled against the package tarball. See API for defaults, React for state/SSR and Theming for required host variables. For richer localized examples use the public examples entrypoint.

Open playground ↗

State machine · Minimal spec

state-machine.tsx
import { useId, useMemo, useState } from 'react'
import { buildAdjacency, connectedIds, diagramEdges, type DiagramSpec } from '@aesthc/diagram-lib'
import { layoutDiagram } from '@aesthc/diagram-lib/layouts'
import { DiagramCanvas } from '@aesthc/diagram-lib/canvas'
import '@aesthc/diagram-lib/styles.css'

const spec = {
  "type": "state-machine",
  "caption": "Job lifecycle",
  "legend": {
    "main": "Main path",
    "branch": "Alternative"
  },
  "states": [
    {
      "id": "a",
      "label": "Pending",
      "initial": true
    },
    {
      "id": "b",
      "label": "Complete",
      "final": true
    }
  ],
  "transitions": [
    {
      "id": "finish",
      "from": "a",
      "to": "b",
      "label": "finish"
    }
  ]
} satisfies DiagramSpec

export function Diagram() {
  const instanceId = useId()
  const [hovered, setHovered] = useState<string | null>(null)
  const [focused, setFocused] = useState<string | null>(null)
  const [selected, setSelected] = useState<string | null>(null)
  const layout = useMemo(() => layoutDiagram(spec), [])
  const adjacency = useMemo(() => buildAdjacency(diagramEdges(spec)), [])
  const active = hovered ?? focused ?? selected
  const highlight = active ? connectedIds(active, adjacency) : null
  return <DiagramCanvas layout={layout} highlight={highlight}
    activeNodeId={active} focusedNodeId={focused} selectedNodeId={selected}
    onTooltipNodeChange={(id, open) => setHovered(open ? id : null)}
    onFocusNode={setFocused} onSelectNode={(id) => setSelected(selected === id ? null : id)}
    onDismissNode={() => { setHovered(null); setSelected(null) }}
    instanceId={instanceId} ariaLabel={spec.caption} nodeVisuals={{}} />
}

Boundaries#

IDs must be unique; references must exist. Parallel relations need explicit IDs when their identity must survive reordering. Large graphs and very long labels need host-specific testing; there is no automatic text measurement or drag editor.
